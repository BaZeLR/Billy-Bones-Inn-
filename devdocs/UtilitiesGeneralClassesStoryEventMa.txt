        def dump_raw_tuple(self, thread_name=None):
            """Print original clean tuple definition for a thread"""
            if not self.raw_thread_list:
                return "No raw data loaded yet"

            result = []
            for thread_item in self.raw_thread_list:
                if not isinstance(thread_item, (list, tuple)):
                    continue
                # thread_item is usually LThreadData / ThreadData instance
                tdata = thread_item
                if hasattr(tdata, "name") and (thread_name is None or thread_name in tdata.name):
                    result.append({
                        "name": getattr(tdata, "name", "unknown"),
                        "level": getattr(tdata, "level", None),
                        "person": getattr(tdata, "person", None),
                        "subname": getattr(tdata, "subname", None),
                        "triggers": getattr(tdata, "triggers", None)  # still contains original event tuples
                    })
            return result

        def get_original_event_tuple(self, target_label):
            """Find the exact original 11-element tuple for any event label"""
            for thread_item in self.raw_thread_list or []:
                for trigger_list in getattr(thread_item, "triggers", []):
                    for evt_tuple in trigger_list:
                        if isinstance(evt_tuple, (list, tuple)) and evt_tuple and evt_tuple[0] == target_label:
                            return evt_tuple  # ← exact original tuple
            return None
