# main_ui_screen.rpy
# Source of truth template.

# Variables provided by game code.
default UI_mode = "scene"              # "scene" | "tavern" | "help" | "mc" | "char"
default UI_char_dropdown = False
default UI_selected_char = ""

default UI_game_date = "Day 1"
default UI_time_of_day = "Morning"
default UI_days_in_game = 1

default UI_chores = {
    "chop_wood": False,
    "bring_woods": False,
    "make_fire": False,
    "hot_water": False,
}

default UI_actions = []                  # Optional extra actions [{"label":..., "action":...}]
default UI_location_chars = []

default UI_char_portrait = ""
default UI_char_desc = ""
default UI_char_clothes_desc = ""
default UI_char_skills_desc = ""
default UI_char_friends = 0

default UI_location_name = ""
default UI_tavern_report_text = ""
default UI_mc_report_text = ""

screen main_ui():
    python:
        _cur_loc = str(globals().get("CurLoc", globals().get("location", "TavernMain")))
        _desc = _mui_room_text()
        _status_lines = _mui_status_lines(_cur_loc)
        _rows = _mui_action_rows(_cur_loc)
        _extra = list(UI_actions) if isinstance(UI_actions, list) else []
        _chars = list(UI_location_chars) if isinstance(UI_location_chars, list) else []
        _textbox_h = int(getattr(gui, "textbox_height", 278))
        _usable_h = max(360, int(config.screen_height) - _textbox_h)

    fixed:
        xfill True
        ysize _usable_h

        hbox:
            xpos 12
            ypos 12
            spacing 12

            vbox:
                xmaximum int((config.screen_width - 36) * 0.72)
                ymaximum _usable_h - 24
                yfill True
                spacing 10
                use main_ui_left_panel(_desc, _cur_loc)

            viewport:
                xmaximum int((config.screen_width - 36) * 0.28)
                ymaximum _usable_h - 24
                yfill True
                draggable True
                mousewheel True

                vbox:
                    spacing 10

                    if _cur_loc == "Intro":
                        textbutton "Приступить к управлению трактиром":
                            text_size 14
                            action Call("intro_start_manage_tavern")
                    else:
                        hbox:
                            spacing 8
                            textbutton "T":
                                text_size 20
                                action [SetVariable("UI_mode", "tavern"), SetVariable("UI_char_dropdown", False)]
                            textbutton "C":
                                text_size 20
                                action ToggleVariable("UI_char_dropdown")
                            textbutton "H":
                                text_size 20
                                action [SetVariable("UI_char_dropdown", False), Function(mui_jump_to, _cur_loc, "TavernHelp", "")]
                            textbutton "S":
                                text_size 20
                                action If(
                                    renpy.has_screen("qsp_time_change_menu"),
                                    [SetVariable("UI_char_dropdown", False), Show("qsp_time_change_menu")],
                                    NullAction(),
                                )
                            textbutton "MC":
                                text_size 20
                                action [SetVariable("UI_mode", "mc"), SetVariable("UI_char_dropdown", False)]

                        if UI_char_dropdown:
                            vbox:
                                spacing 6
                                textbutton "Sandra":
                                    text_size 20
                                    action [SetVariable("UI_selected_char", "Sandra"), SetVariable("UI_mode", "char"), SetVariable("UI_char_dropdown", False)]
                                textbutton "Melissa":
                                    text_size 20
                                    action [SetVariable("UI_selected_char", "Melissa"), SetVariable("UI_mode", "char"), SetVariable("UI_char_dropdown", False)]
                                textbutton "Amanda":
                                    text_size 20
                                    action [SetVariable("UI_selected_char", "Amanda"), SetVariable("UI_mode", "char"), SetVariable("UI_char_dropdown", False)]

                        vbox:
                            spacing 4
                            if UI_location_name:
                                text UI_location_name size 20
                            text "Date: [UI_game_date]" size 20
                            text "Time: [UI_time_of_day]" size 20
                            text "Days in game: [UI_days_in_game]" size 20
                            for _line in _status_lines:
                                text _line size 20

                        vbox:
                            spacing 6
                            text "Chores" size 20
                            hbox:
                                spacing 8
                                textbutton "chop wood":
                                    text_size 20
                                    action ToggleDict(UI_chores, "chop_wood")
                                text ("OK" if UI_chores["chop_wood"] else "...") size 20
                            hbox:
                                spacing 8
                                textbutton "bring woods":
                                    text_size 20
                                    action ToggleDict(UI_chores, "bring_woods")
                                text ("OK" if UI_chores["bring_woods"] else "...") size 20
                            hbox:
                                spacing 8
                                textbutton "make fire":
                                    text_size 20
                                    action ToggleDict(UI_chores, "make_fire")
                                text ("OK" if UI_chores["make_fire"] else "...") size 20
                            hbox:
                                spacing 8
                                textbutton "hot water":
                                    text_size 20
                                    action ToggleDict(UI_chores, "hot_water")
                                text ("OK" if UI_chores["hot_water"] else "...") size 20

                        text "Actions / Navigation" size 20
                        viewport:
                            draggable True
                            mousewheel True
                            ymaximum 280
                            yfill True
                            vbox:
                                spacing 6
                                if _extra:
                                    for a in _extra:
                                        if isinstance(a, dict) and a.get("label", "") and a.get("action", None):
                                            textbutton a["label"]:
                                                text_size 20
                                                action a["action"]
                                else:
                                    for _row in _rows:
                                        if _row.get("kind", "") == "nav":
                                            textbutton _row.get("label", ""):
                                                text_size 20
                                                action Function(mui_jump_to, _cur_loc, _row.get("target", ""), _row.get("jump_label", ""))
                                        elif _row.get("kind", "") == "call_label":
                                            textbutton _row.get("label", ""):
                                                text_size 20
                                                action Call(_row.get("value", ""))
                                        elif _row.get("kind", "") == "show_screen":
                                            if _row.get("value", "") == "menu_tavernstat_popup":
                                                textbutton _row.get("label", ""):
                                                    text_size 20
                                                    action Show("menu_tavernstat_popup", from_status=True)
                                            else:
                                                textbutton _row.get("label", ""):
                                                    text_size 20
                                                    action Show(_row.get("value", ""))
                                        elif _row.get("kind", "") == "select_object":
                                            textbutton _row.get("label", ""):
                                                text_size 20
                                                action Function(qsp_select_object, _row.get("value", ""))
                                    if len(_rows) == 0:
                                        text "No actions." size 20

                        vbox:
                            spacing 6
                            text "In this location" size 20
                            if _chars:
                                for cname in _chars:
                                    textbutton cname:
                                        text_size 20
                                        action [SetVariable("UI_selected_char", cname), SetVariable("UI_mode", "char"), SetVariable("UI_char_dropdown", False)]
                            else:
                                text "Nobody here." size 20

screen main_ui_left_panel(desc, cur_loc):
    if UI_mode == "scene":
        vbox:
            xfill True
            yfill True
            spacing 8

            if str(cur_loc) == "Intro":
                viewport:
                    xfill True
                    yfill True
                    draggable True
                    mousewheel True
                    text desc size 14
            else:
                null:
                    xfill True
                    yfill True

    elif UI_mode == "tavern":
        vbox:
            xfill True
            yfill True
            spacing 10
            text "Tavern Report" size 24
            viewport:
                draggable True
                mousewheel True
                yfill True
                if UI_tavern_report_text:
                    text UI_tavern_report_text size 20
                else:
                    text "No tavern report data." size 20
            textbutton "Close":
                text_size 20
                action SetVariable("UI_mode", "scene")

    elif UI_mode == "help":
        vbox:
            xfill True
            yfill True
            spacing 10
            text "Help"
            text "Render help / info text here."
            textbutton "Back to Location":
                action SetVariable("UI_mode", "scene")

    elif UI_mode == "mc":
        vbox:
            xfill True
            yfill True
            spacing 10
            text "MC Report" size 24
            viewport:
                draggable True
                mousewheel True
                yfill True
                if UI_mc_report_text:
                    text UI_mc_report_text size 20
                else:
                    text "No player report data." size 20
            textbutton "Close":
                text_size 20
                action SetVariable("UI_mode", "scene")

    elif UI_mode == "char":
        use main_ui_character_card()

screen main_ui_character_card():
    vbox:
        xfill True
        yfill True
        spacing 12

        text "[UI_selected_char]" size 34

        hbox:
            spacing 16

            fixed:
                xsize 220
                ysize 330
                if UI_char_portrait:
                    add UI_char_portrait
                else:
                    text "Portrait\n200x300"

            vbox:
                spacing 10
                text "Friends: [UI_char_friends]"

                vbox:
                    spacing 6
                    text "Current description:"
                    text UI_char_desc

                vbox:
                    spacing 6
                    text "Clothes worn:"
                    text UI_char_clothes_desc

                vbox:
                    spacing 6
                    text "Skills:"
                    text UI_char_skills_desc

        textbutton "Back to Location":
            action SetVariable("UI_mode", "scene")
